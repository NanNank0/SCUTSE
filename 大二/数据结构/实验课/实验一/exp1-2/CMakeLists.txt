cmake_minimum_required(VERSION 3.21)
project(exp1_2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_add_executable(exp1_2 WIN32 main.cpp)
target_link_libraries(exp1_2 PRIVATE Qt6::Widgets)
if(MSVC)
  target_compile_options(exp1_2 PRIVATE /utf-8)
endif()
