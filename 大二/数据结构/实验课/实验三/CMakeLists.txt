cmake_minimum_required(VERSION 3.21)
project(DataStructuresLab3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_add_executable(SortCompare WIN32 compare.cpp)
target_link_libraries(SortCompare PRIVATE Qt6::Widgets)
add_subdirectory(InsertSort)
add_subdirectory(BubbleSort)
add_subdirectory(SelectionSort)
add_subdirectory(MergeSort)
add_subdirectory(QuickSort)
add_subdirectory(HeapSort)
add_subdirectory(RadixSort)

enable_testing()
add_executable(sort_tests tests/sort_tests.cpp)
add_test(NAME sorting_correctness COMMAND sort_tests)

file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/测试数据" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  add_compile_options(/utf-8)
endif()

add_executable(ui_tests tests/ui_tests.cpp)
target_link_libraries(ui_tests PRIVATE Qt6::Widgets)
add_test(NAME gui_interactions COMMAND ui_tests)
set_tests_properties(gui_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
