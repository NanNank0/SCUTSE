cmake_minimum_required(VERSION 3.21)
project(DataStructuresLab4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
add_subdirectory(exp4-1)
add_subdirectory(exp4-2)
add_subdirectory(exp4-3)
enable_testing()
add_executable(graph_tests tests/graph_tests.cpp)
target_link_libraries(graph_tests PRIVATE Qt6::Widgets)
add_test(NAME graph_correctness COMMAND graph_tests)

file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/测试数据" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  add_compile_options(/utf-8)
endif()

add_executable(ui_tests tests/ui_tests.cpp)
target_link_libraries(ui_tests PRIVATE Qt6::Widgets)
add_test(NAME gui_interactions COMMAND ui_tests)
set_tests_properties(gui_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
